Suggested implementation
=======================

Create one common base page and inherit all protected ASPX pages from it. This removes the authorization call from every page and avoids relative login paths.

AuthorizedPage.cs
-----------------

using System;
using System.Data;
using System.Web;
using System.Web.UI;

public abstract class AuthorizedPage : Page
{
    protected override void OnInit(EventArgs e)
    {
        base.OnInit(e);

        if (!IsAuthorizedUser())
        {
            Response.Redirect(
                VirtualPathUtility.ToAbsolute("~/login.aspx"),
                false);

            Context.ApplicationInstance.CompleteRequest();
        }
    }

    private bool IsAuthorizedUser()
    {
        string userName = Convert.ToString(Session["USERID"]);

        if (string.IsNullOrWhiteSpace(userName))
        {
            return false;
        }

        string pageName = GetFileName(Request.Path);
        Session["pageName"] = pageName;

        string queryStringKey = Request.QueryString.AllKeys.Length > 0
            ? Request.QueryString.AllKeys[0]
            : string.Empty;

        string sql = @"
            SELECT MNU.Menu_filename AS FILENAME,
                   MNU.Menu_Role AS ROLE,
                   RLE.Status AS STATUS,
                   USR.loginid AS LOGINID
            FROM FRM_MENU MNU
            INNER JOIN FRM_USERROLE RLE
                ON MNU.Menu_Role = RLE.USER_ROLE
            INNER JOIN FRM_USER USR
                ON RLE.USER_USERID = USR.LoginID
            WHERE UPPER(MNU.Menu_filename) LIKE :PAGE_NAME
              AND RLE.Status = 'A'
              AND MNU.Status = 'A'
              AND USR.Status = 'A'
              AND UPPER(USR.loginid) = :USER_ID";

        // Pass PAGE_NAME and USER_ID as parameters using the existing database helper.
        DataSet accessData = ExecuteQuery(sql, new
        {
            PAGE_NAME = "%" + pageName.ToUpperInvariant() + "%",
            USER_ID = userName.ToUpperInvariant()
        });

        if (accessData == null || accessData.Tables.Count == 0)
        {
            return false;
        }

        foreach (DataRow row in accessData.Tables[0].Rows)
        {
            string configuredPath = Convert.ToString(row["FILENAME"]);

            if (string.Equals(
                    GetFileName(configuredPath),
                    pageName,
                    StringComparison.OrdinalIgnoreCase))
            {
                return true;
            }
        }

        return IsAnonymousPage(queryStringKey);
    }

    private static string GetFileName(string path)
    {
        if (string.IsNullOrWhiteSpace(path))
        {
            return string.Empty;
        }

        string cleanPath = HttpUtility.UrlDecode(path)
            .Split('?')[0]
            .TrimEnd('/');

        int lastSlash = cleanPath.LastIndexOf('/');

        return lastSlash >= 0
            ? cleanPath.Substring(lastSlash + 1)
            : cleanPath;
    }

    private bool IsAnonymousPage(string queryStringKey)
    {
        return GetFileName(Request.Path).EndsWith(
                   ".asmx",
                   StringComparison.OrdinalIgnoreCase)
               || IsQueryStringConfigured(queryStringKey);
    }

    private bool IsQueryStringConfigured(string queryStringKey)
    {
        const string sql = @"
            SELECT 1
            FROM frm_gnarr
            WHERE Code = 'Qry_str'
              AND Status = 'A'
              AND Value = :QUERY_STRING_KEY";

        DataSet result = ExecuteQuery(sql, new
        {
            QUERY_STRING_KEY = queryStringKey
        });

        return result != null
               && result.Tables.Count > 0
               && result.Tables[0].Rows.Count > 0;
    }
