public static void ValidatePageAccess()
{
    string referrer = HttpContext.Current.Request.UrlReferrer != null
        ? HttpContext.Current.Request.UrlReferrer.AbsolutePath
        : string.Empty;

    if (string.IsNullOrEmpty(referrer) ||
        !referrer.EndsWith("/default.aspx", StringComparison.OrdinalIgnoreCase))
    {
        HttpContext.Current.Response.Redirect("../default.aspx", false);
        HttpContext.Current.Context.ApplicationInstance.CompleteRequest();
    }
}

protected void Page_Load(object sender, EventArgs e)
{
    Common.ValidatePageAccess();
