string tempFolder = Server.MapPath("~/TEMP/");

// Delete previously generated PDF for this user
if (Session["GeneratedPDF"] != null)
{
    string oldFileName = Session["GeneratedPDF"].ToString();
    string oldFilePath = Path.Combine(tempFolder, oldFileName);

    if (File.Exists(oldFilePath))
    {
        File.Delete(oldFilePath);
    }
}

// Generate unpredictable filename
string newFileName = Guid.NewGuid().ToString("N") + ".pdf";
string newFilePath = Path.Combine(tempFolder, newFileName);

// Create PDF
PdfWriter.GetInstance(
    _Document,
    new FileStream(newFilePath, FileMode.Create)
);

// Store the new filename in Session
Session["GeneratedPDF"] = newFileName;

// URL for the newly generated PDF
string pdfUrl = RQ.getAppSettings("URL") + "/TEMP/" + newFileName;

__string += "viewASPFile(\"" + pdfUrl + "\");";

string tempFolder = Server.MapPath("~/TEMP/");
string userId = Session["USERID"].ToString();

// Delete previous PDF files for this user
string[] oldFiles = Directory.GetFiles(
    tempFolder,
    userId + "_*.pdf"
);

foreach (string file in oldFiles)
{
    File.Delete(file);
}

// Generate unpredictable filename
string fileName = userId + "_" + Guid.NewGuid().ToString("N") + ".pdf";
string filePath = Path.Combine(tempFolder, fileName);

// Create PDF
PdfWriter.GetInstance(
    _Document,
    new FileStream(filePath, FileMode.Create)
);

// URL of newly generated PDF
string pdfUrl = RQ.getAppSettings("URL") + "/TEMP/" + fileName;

__string += "viewASPFile(\"" + pdfUrl + "\");";
